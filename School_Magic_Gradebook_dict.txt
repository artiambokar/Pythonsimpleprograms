student_dict = {'Anay': [50, 60, 70], 'Kiara':[60,70,89], 'Sayra':[56,89,67]}

def add_grade(student_name, grade):
  if student_name in student_dict:
    student_dict  [student_name].append(grade)
  else:
    student_dict[student_name] = [grade]
    print(f"Student '{student_name}' added with grade {grade}.")
    
def update_grade(student_name, new_grade):
  if student_name in student_dict:
    student_dict[student_name][-1] = new_grade
    print(f"Grade for student '{student_name}' updated to {new_grade}.")
    #student_dict[student_name].append(new_grade)
    print(f"Student '{student_name}' not found.")

def calculate_average(student_name):
  if student_name in student_dict:
    grades = student_dict[student_name]
    average_grade = sum(grades) / len(grades)
    return average_grade

def get_grade_letter(average_grade):
  if average_grade > 90:
    return 'A+'
  elif average_grade>80 and average_grade<=90:
    return 'A'
  elif average_grade>70 and average_grade<=80:
    return 'B+'
  elif average_grade>60 and average_grade<=70:
    return 'B'
  elif average_grade>50 and average_grade<=60:
    return 'C+'
  elif average_grade>40 and average_grade<=50:
    return 'C'
  elif average_grade>30 and average_grade<=40:
    return 'D+'
  elif average_grade>20 and average_grade<=30:
    return 'D'
  else:
    return 'E'

def view_summary():
  for student, grades in student_dict.items():
    grade = calculate_average(student)
    print(f"Student: {student}")
    print(f"Grades: {', '.join(map(str, grades))}")
    print(f"Average: {grade}")
    print(f"Grade Letter: {get_grade_letter(grade)}")
    print()
